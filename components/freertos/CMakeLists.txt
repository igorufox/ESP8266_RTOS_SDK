# ESPurna: FreeRTOS-Kernel V11 of the one kernel package (framework-freertos-kernel, FREERTOS_KERNEL_DIR given by
# scripts/esp8266_rtos_sdk.py), the ESP8266 port of the SDK (port/esp8266). The SDK's own 10.0.1 kernel is gone: the
# three chips run the same kernel source. Espressif's additions to the kernel are in the port, through the kernel's own
# hooks (freertos_tasks_c_additions.h, portCLEAN_UP_TCB), the kernel files themselves stay unchanged.
if(NOT FREERTOS_KERNEL_DIR)
    # the requirements pass of the build runs this file in a script of its own, without the cache: the environment
    set(FREERTOS_KERNEL_DIR "$ENV{FREERTOS_KERNEL_DIR}")
endif()
if(NOT FREERTOS_KERNEL_DIR AND NOT CMAKE_BUILD_EARLY_EXPANSION)
    message(FATAL_ERROR "FREERTOS_KERNEL_DIR: the FreeRTOS kernel package (framework-freertos-kernel) is not given")
endif()

# <freertos/task.h> ...: the SDK's code includes the kernel headers under freertos/, the kernel package has them at its
# include/: forwarding headers in the build dir
set(forward_dir "${CMAKE_BINARY_DIR}/freertos_kernel_include")
file(GLOB kernel_headers "${FREERTOS_KERNEL_DIR}/include/*.h")
foreach(header ${kernel_headers})
    get_filename_component(name "${header}" NAME)
    set(text "/* generated by components/freertos/CMakeLists.txt */\n#include \"${header}\"\n")
    set(out "${forward_dir}/freertos/${name}")
    if(EXISTS "${out}")
        file(READ "${out}" old)
    else()
        set(old "")
    endif()
    if(NOT old STREQUAL text)
        file(WRITE "${out}" "${text}")
    endif()
endforeach()

set(kernel_srcs
    "${FREERTOS_KERNEL_DIR}/event_groups.c"
    "${FREERTOS_KERNEL_DIR}/list.c"
    "${FREERTOS_KERNEL_DIR}/queue.c"
    "${FREERTOS_KERNEL_DIR}/stream_buffer.c"
    "${FREERTOS_KERNEL_DIR}/tasks.c"
    "${FREERTOS_KERNEL_DIR}/timers.c")

set(port_srcs
    "port/esp8266/freertos_hooks.c"
    "port/esp8266/os_cpu_a.S"
    "port/esp8266/panic.c"
    "port/esp8266/port.c"
    "port/esp8266/xtensa_context.S"
    "port/esp8266/xtensa_vectors.S")

idf_component_register(SRCS ${kernel_srcs} ${port_srcs}
                       INCLUDE_DIRS "${forward_dir}" "${FREERTOS_KERNEL_DIR}/include"
                                    "port/esp8266/include" "port/esp8266/include/freertos"
                       LDFRAGMENTS "linker.lf")

target_link_libraries(${COMPONENT_LIB} INTERFACE "-Wl,--undefined=uxTopUsedPriority")

# the kernel's own files: the port's additions to tasks.c (freertos_tasks_c_additions.h) see the kernel's internals
set_source_files_properties(${kernel_srcs} PROPERTIES COMPILE_DEFINITIONS _ESP_FREERTOS_INTERNAL)
